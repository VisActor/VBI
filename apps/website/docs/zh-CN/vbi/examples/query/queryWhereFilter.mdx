## equal-filter

先筛选华东地区的源记录，再按品类汇总销售额。


> 标签: `query` `whereFilter`


```tsx preview
import { VBI, type VBIQueryBuilder, type VBIQueryDSL } from '@visactor/vbi'
import { QueryResultRender } from '@components'
import { useEffect, useState } from 'react'

export default () => {
  const [result, setResult] = useState<{ value: unknown; dsl: VBIQueryDSL } | null>(null)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    let disposed = false
    const run = async () => {
      const builder = VBI.query.create({
        ...VBI.query.createEmpty('demoSupermarket'),
        ...{
          "dimensions": [
            {
              "id": "productType",
              "field": "product_type",
              "alias": "品类"
            }
          ],
          "measures": [
            {
              "id": "sales",
              "field": "sales",
              "alias": "销售额",
              "aggregate": {
                "func": "sum"
              }
            }
          ],
          "limit": 3
        }
      })
      try {
        const applyBuilder = (builder: VBIQueryBuilder) => {
  builder.whereFilter.add('area', node => node.setOperator('eq').setValue('华东'))
}
        const value = await applyBuilder(builder) ?? await builder.query()
        if (!disposed) setResult({ value, dsl: builder.build() })
      } catch (err) {
        if (!disposed) setError(err instanceof Error ? err.message : String(err))
      } finally {
        builder.doc.destroy()
      }
    }
    void run()
    return () => { disposed = true }
  }, [])

  if (error) return <div role='alert'>{error}</div>
  if (result === null) return <div>Loading...</div>

  return <QueryResultRender value={result.value} dsl={result.dsl} />
}
```

## nested-filters

以 AND 连接华东地区条件与品类 OR 分组，统计华东办公用品或家具的销售额。


> 标签: `query` `whereFilter`


```tsx preview
import { VBI, type VBIQueryBuilder, type VBIQueryDSL } from '@visactor/vbi'
import { QueryResultRender } from '@components'
import { useEffect, useState } from 'react'

export default () => {
  const [result, setResult] = useState<{ value: unknown; dsl: VBIQueryDSL } | null>(null)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    let disposed = false
    const run = async () => {
      const builder = VBI.query.create({
        ...VBI.query.createEmpty('demoSupermarket'),
        ...{
          "dimensions": [
            {
              "id": "productType",
              "field": "product_type",
              "alias": "品类"
            }
          ],
          "measures": [
            {
              "id": "sales",
              "field": "sales",
              "alias": "销售额",
              "aggregate": {
                "func": "sum"
              }
            }
          ],
          "limit": 3
        }
      })
      try {
        const applyBuilder = (builder: VBIQueryBuilder) => {
  builder.whereFilter
    .add('area', node => node.setOperator('eq').setValue('华东'))
    .addGroup('or', group => {
      group.add('product_type', node => node.setOperator('eq').setValue('办公用品'))
      group.add('product_type', node => node.setOperator('eq').setValue('家具'))
    })
}
        const value = await applyBuilder(builder) ?? await builder.query()
        if (!disposed) setResult({ value, dsl: builder.build() })
      } catch (err) {
        if (!disposed) setError(err instanceof Error ? err.message : String(err))
      } finally {
        builder.doc.destroy()
      }
    }
    void run()
    return () => { disposed = true }
  }, [])

  if (error) return <div role='alert'>{error}</div>
  if (result === null) return <div>Loading...</div>

  return <QueryResultRender value={result.value} dsl={result.dsl} />
}
```
