# PeriodComparison

先按明确的本期与基期汇总，再计算差额和增长率。下面依次展示环比、同月同比和累计同比；基期为 0 或缺失时，增长率返回 NULL。

## month-over-month

比较 2026-03 与 2026-02 的地区销售额：先按月分桶并汇总，再计算差额和环比。4 月数据排除；基期为 0 时环比为 NULL，缺失期间不补零，实际销售额为 0 则保留，0.2 表示增长 20%。

  "schema": [
    {
      "name": "region",
      "type": "string"
    },

    {
      "region": "B",
      "orderDate": "2026-03-10 10:00:00",
      "sales": 80
    },
    {
      "region": "C",
      "orderDate": "2026-02-10 10:00:00",
      "sales": 0
    },
    {
      "region": "C",
      "orderDate": "2026-03-10 10:00:00",
      "sales": 20
    },
    {
      "region": "D",
      "orderDate": "2026-03-10 10:00:00",
      "sales": 50
    },
    {
      "region": "E",
      "orderDate": "2026-02-10 10:00:00",
      "sales": 60
    },
    {
      "region": "E",
      "orderDate": "2026-03-10 10:00:00",
      "sales": 0
    },
    {
      "region": "F",
      "orderDate": "2026-02-10 10:00:00",
      "sales": 80
    }
  ],
  "vquery": {
    "from": {
      "from": {
        "select": [
          "region",
          {
            "field": "orderDate",
            "alias": "month",
            "aggr": {
              "func": "to_month"
            }
          },
          "sales"
        ]
      },
      "select": [
        "region",
        {
          "field": "sales",
          "alias": "currentSales",
          "aggr": {
            "func": "sum"
          },
          "when": {
            "op": "and",
            "conditions": [
              {
                "field": "month",
                "op": "=",
                "value": "2026-03"
              }
            ]
          }
        },
        {
          "field": "sales",
          "alias": "baselineSales",
          "aggr": {
            "func": "sum"
          },
          "when": {
            "op": "and",
            "conditions": [
              {
                "field": "month",
                "op": "=",
                "value": "2026-02"
              }
            ]
          }
        }
      ],
      "where": {
        "op": "and",
        "conditions": [
          {
            "field": "month",
            "op": "in",
            "value": [
              "2026-02",
              "2026-03"
            ]
          }
        ]
      },
      "groupBy": [
        "region"
      ]
    },
    "select": [
      "region",
      "currentSales",
      "baselineSales",
      {
        "expr": "currentSales-baselineSales",
        "alias": "delta"
      },
      {
        "expr": "(currentSales-baselineSales)/NULLIF(baselineSales,0)",
        "alias": "growthRate"
      }
    ],
    "orderBy": [
      {
        "field": "region",
        "order": "asc"
      }
    ]
  }
}

  return <VQueryResultRender vqueryConfig={vqueryConfig} />
}
