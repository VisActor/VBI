# PivotTable

## Basic PivotTable

A pivot table for summarizing and analyzing large datasets.

```tsx preview

export default () => {
  const vseedConfig = {
  "chartType": "pivotTable",
  "dataset": [
    {
      "date": "2019",
      "type": "A",
      "goalProfit": 10,
      "profit": 10,
      "sales": 100,
      "leftCount": 1,
      "salesCount": 100
    },
    {
      "date": "2020",
      "type": "A",
      "goalProfit": 30,
      "profit": 30,
      "sales": 320,
      "leftCount": 2,
      "salesCount": 200
    },
    {
      "date": "2021",
      "type": "A",
      "goalProfit": 30,
      "profit": 30,
      "sales": 300,
      "leftCount": 3,
      "salesCount": 300
    },
    {
      "date": "2022",
      "type": "A",
      "goalProfit": 50,
      "profit": 50,
      "sales": 240,
      "leftCount": 4,
      "salesCount": 400
    },
    {
      "date": "2023",
      "type": "A",
      "goalProfit": 40,
      "profit": 40,
      "sales": 500,

      "salesCount": 500

  return <VSeedRender vseed={vseedConfig} />
```

## 透视表布局与字段列宽

行维度、列维度和指标分别设置列宽，表格保持固定行高并自动填满容器。

```tsx preview
import { VSeedRender } from '@components'

export default () => {
  const vseedConfig = {
  "chartType": "pivotTable",
  "dataset": [
    {
      "region": "华东 East",
      "year": "2025",
      "sales": 1200
    },
    {
      "region": "华东 East",
      "year": "2026",
      "sales": 1600
    },
    {
      "region": "华南 South",
      "year": "2025",
      "sales": 900
    },
    {
      "region": "华南 South",
      "year": "2026",
      "sales": 1300
    }
  ],
  "dimensions": [
    {
      "id": "region",
      "alias": "区域",
      "encoding": "row",
      "width": 150
    },
    {
      "id": "year",
      "alias": "年份",
      "encoding": "column",
      "width": 120
    }
  ],
  "measures": [
    {
      "id": "sales",
      "alias": "销售额",
      "width": 130
    }
  ],
  "indicatorsAsCol": true,
  "heightMode": "standard",
  "rowHeight": 38,
  "headerRowHeight": 44,
  "autoWrapText": false,
  "autoFillWidth": true
}

  return <VSeedRender vseed={vseedConfig} />
}
```
