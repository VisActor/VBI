# Animation

## barLikeAnimation

Bar-like charts support appear, update, and loop animation.

```tsx preview
import { VSeedRender } from '@components'

export default () => {
  const vseedConfig = {
    chartType: 'column',
    animation: {
      enable: true,
      params: {
        appear: {
          enable: true,
          effects: ['growth'],
          duration: 800,
          ease: 'cubicOut',
        },
        update: {
          enable: true,
          effects: ['moveIn'],
          duration: 600,
          ease: 'quadOut',
        },
        loop: {
          enable: true,
          interval: 1200,
          loop: {
            enable: true,
            effects: ['highLight'],
            duration: 1000,
          },
          atmosphere: {
            effect: 'ripple',
            color: '#4e79a7',
            ease: 'linear',
          },
        },
      },
    },
    dataset: [
      {
        date: '2019',
        sales: 20,
      },
      {
        date: '2020',
        sales: 60,
      },
      {
        date: '2021',
        sales: 45,
      },
    ],
    dimensions: [
      {
        id: 'date',
        alias: 'Date',
      },
    ],
    measures: [
      {
        id: 'sales',
        alias: 'Sales',
      },
    ],
  }

  return <VSeedRender vseed={vseedConfig} />
}
```

## Data update timing

Data enter, update and exit share timing without requiring an effect name. Durations and loop intervals use milliseconds.

```tsx preview
import { VSeedRender } from '@components'

export default () => {
  const vseedConfig = {
    chartType: 'area',
    dataset: [
      {
        date: '2019',

        value: 10,
      },
      {
        date: '2020',
        region: 'East',
        value: 20,
      },
      {
        date: '2021',
        region: 'East',
        value: 14,
      },
      {
        date: '2022',
        region: 'East',
        value: 30,
      },
      {
        date: '2019',
        region: 'West',
        value: 5,
      },
      {
        date: '2020',
        region: 'West',
        value: 10,
      },
      {
        date: '2021',
        region: 'West',
        value: 8,
      },
      {
        date: '2022',
        region: 'West',
        value: 17,
      },
    ],
    dimensions: [
      {
        id: 'date',
      },
      {
        id: 'region',
        encoding: 'color',
      },
    ],
    measures: [
      {
        id: 'value',
      },
    ],
    pointStyle: {
      pointVisible: false,
    },
    lineStyle: {
      lineSmooth: true,
    },

      enable: false,
    },

          enable: false,

        },
      },
    },
  }

  return <VSeedRender vseed={vseedConfig} />
```
