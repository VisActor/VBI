# DualAxis

## Basic DualAxis

A dual-axis chart combining two different chart types or scales.

```tsx preview 
import { VSeedRender } from '@components'

export default () => {
  const vseedConfig = {
  "chartType": "dualAxis",
  "dataset": [
    {
      "date": "2019",
      "profit": 20,
      "sales": 100
    },
    {
      "date": "2020",
      "profit": 30,
      "sales": 600
    },
    {
      "date": "2021",
      "profit": 30,
      "sales": 600
    },
    {
      "date": "2022",
      "profit": 50,
      "sales": 500
    },
    {
      "date": "2023",
      "profit": 10,
      "sales": 800
    }
  ],
  "measures": [
    {
      "id": "sales",
      "alias": "销售额"
    },
    {
      "id": "profit",
      "alias": "利润"
    }
  ]
}

  return <VSeedRender vseed={vseedConfig} />
}
```
