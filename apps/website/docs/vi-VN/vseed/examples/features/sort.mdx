# Sort

## Truyền thuyết tự sắp xếp

Truyền thuyết tự sắp xếp

```tsx preview 
import { VSeedRender } from '@components'

export default () => {
  const vseedConfig = {
  "chartType": "column",
  "dataset": [
    {
      "date": "2019",
      "profit": 10,
      "sales": 20,
      "count": 15
    },
    {
      "date": "2020",
      "profit": 30,
      "sales": 60,
      "count": 33
    },
    {
      "date": "2021",
      "profit": 30,
      "sales": 60,
      "count": 22
    },
    {
      "date": "2022",
      "profit": 50,
      "sales": 100,
      "count": 14
    },
    {
      "date": "2023",
      "profit": 40,
      "sales": 80,
      "count": 30
    }
  ],
  "dimensions": [
    {
      "id": "date",
      "alias": "日期"
    }
  ],
  "measures": [
    {
      "id": "count",
      "alias": "数量"
    },
    {
      "id": "sales",
      "alias": "销售额"
    },
    {
      "id": "profit",
      "alias": "利润"
    }
  ],
  "sortLegend": {
    "order": "desc"
  }
}

  return <VSeedRender vseed={vseedConfig} />
}
```
